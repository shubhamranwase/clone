import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

# Read CSV File
data = pd.read_csv("student.csv")

# Input and Output
x = data[["Study Hours", "Attendance", "Assignments", "Previous Marks"]]
y = data["Pass"]

# Split Data
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=42)

# Create model
model = LogisticRegression()

# Train model
model.fit(x_train, y_train)

y_pred = model.predict(x_test)

# Accuracy
print("Accuracy:", accuracy_score(y_test, y_pred))

# Predict new student
new_student = pd.DataFrame(
    [[5, 75, 5, 45]], # Placeholder values for the obscured data
    columns=["Study Hours", "Attendance", "Assignments", "Previous Marks"]
)

prediction = model.predict(new_student)
probability = model.predict_proba(new_student)

print("Prediction =", prediction)
print("Probability =", probability)

# ---------- Graph ----------

hours = np.linspace(1, 9, 200)

graph_data = pd.DataFrame({
    "Study Hours": hours,
    "Attendance": np.repeat(data["Attendance"].mean(), 200),
    "Assignments": np.repeat(data["Assignments"].mean(), 200),
    "Previous Marks": np.repeat(data["Previous Marks"].mean(), 200)
})

prob = model.predict_proba(graph_data)[:, 1]

plt.figure(figsize=(8,5))

plt.scatter(data["Study Hours"], data["Pass"], color="blue", s=80, label="Actual Data")
plt.plot(hours, prob, color="red", linewidth=3, label="Logistic Curve")

plt.xlabel("Study Hours")
plt.ylabel("Probability of Pass")
plt.title("Logistic Regression on Non-Separable Data")
plt.grid(True)
plt.legend()

plt.show()
